cmake_minimum_required(VERSION 3.16)
project(cricket_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(cricket_engine STATIC CricketPhysics.cpp MatchSimulation.cpp BowlingAI.cpp CricketCAPI.cpp)
target_include_directories(cricket_engine PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>)
if(NOT EMSCRIPTEN)
 add_executable(cricket_engine_test test_main.cpp)
 target_link_libraries(cricket_engine_test PRIVATE cricket_engine)
endif()
